    82→    82→def detect_country(text):
    83→    83→    """Detect country from text using word-boundary regex (Persish-aware)."""
    84→    84→    if not text:
    85→    85→        return "Unknown"
    86→    86→    # Normalize: collapse whitespace (including newlines) to single spaces for matching
    87→    87→    norm = re.sub(r'\s+', ' ', text)
    88→    88→    # Order by length descending so multi-word matches win
    89→    89→    items = sorted(COUNTRY_FA_EN.items(), key=lambda x: -len(x[0]))
    90→    90→    for fa_name, en_name in items:
    91→    91→        # Word boundary: preceding char must not be Persian letter, following char must not be Persian letter
    92→    92→        escaped = re.escape(fa_name)
    93→    93→        pattern = r'(^|[^\u0600-\u06FF])' + escaped + r'(?=[^\u0600-\u06FF]|$)'
    94→    94→        if re.search(pattern, norm):
    95→    95→            return en_name
    96→    96→    # Fallback: also match "آمریک" prefix (text may be broken as "آمریک ایی")
    97→    97→    if re.search(r'(^|[^\u0600-\u06FF])آمریک(?=[\s\u0600-\u06FF]|$)', norm):
    98→    98→        return "United States"
    99→    99→    # Fallback: check network names

   102→   102→        return 'United Kingdom'
   103→   103→    if 'MTV' in net_text or 'ABC' in net_text or 'TLC' in net_text or 'USA Network' in net_text or 'Up TV' in net_text:
   104→   104→        return 'United States'
   105→   105→    if 'BBC' in net_text:
   106→   106→        return 'United Kingdom'
   107→   107→    if 'Discovery' in net_text:
   108→   108→        return 'United States'
   109→   109→    if 'Netflix' in net_text or 'نتفلیکس' in net_text:
   110→   110→        return 'United States'

   246→   246→def parse_pdf2():
   247→   247→    out = []
   248→   248→    pages = PDF_TEXT[2]
   249→   249→    # Sort programs by page so we can compute end_page = next_program's start_page - 1 (same page if multiple on one page)
   250→   250→    sorted_progs = sorted(PDF2_PROGRAMS, key=lambda x: (x[1],))
   251→   251→    for i, (name, sp) in enumerate(sorted_progs):
   252→   252→        # Determine end page: next program's start page (same or later). For programs on the same page,
   253→   253→        # we still split the text at the next program's name.
   254→   254→        if i + 1 < len(sorted_progs):
   255→   255→            next_name, next_sp = sorted_progs[i + 1]
   256→   256→            ep = next_sp  # may equal sp if same page
   257→   257→        else:
   258→   258→            ep = len(pages) - 1  # last program goes to end
   259→   259→        # Combine text from sp..ep
   260→   260→        text_parts = []
   261→   261→        for p in range(sp, min(ep + 1, len(pages))):
   262→   262→            text_parts.append(pages[p]["text"])
   263→   263→        full = "\n".join(text_parts)
   264→   264→        # Slice description: from after the program name to before the next program name (or end)
   265→   265→        # Find the position of the program name in the text
   266→   266→        # The English name may be on its own line; find the first occurrence
   267→   267→        idx = full.find(name)
   268→   268→        if idx >= 0:
   269→   269→            # Start after the name
   270→   270→            after_name = full[idx + len(name):]
   271→   271→            # If multiple programs on same page, find next English name in after_name
   272→   272→            if i + 1 < len(sorted_progs):
   273→   273→                next_name, _ = sorted_progs[i + 1]
   274→   274→                nidx = after_name.find(next_name)
   275→   275→                if nidx >= 0:
   276→   276→                    after_name = after_name[:nidx]
   277→   277→            desc = after_name
   278→   278→        else:
   279→   279→            desc = full

   282→   282→        desc = clean_text(desc)
   283→   283→        if len(desc) > 2500:
   284→   284→            desc = desc[:2500] + "…"
   285→   285→        # Country detection
   286→   286→        country = detect_country(full)

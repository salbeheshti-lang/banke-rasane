     7→// Read the real app logo (PNG) once at module load — embedded as a base64
     8→// data URL in the PDF cover so it renders in the print-to-PDF window (which
     9→// has no base URL to fetch assets from).
    10→const LOGO_PNG_PATH = resolve(process.cwd(), 'public/banke-rasane-logo.png');
    11→let LOGO_DATA_URL = '';
    12→try {
    13→  const buf = readFileSync(LOGO_PNG_PATH);
    14→  LOGO_DATA_URL = `data:image/png;base64,${buf.toString('base64')}`;
    15→} catch {
    16→  // Fallback: empty string — the cover will just show the brand text.
    17→  LOGO_DATA_URL = '';
    18→}
    19→
    20→export async function POST(request: NextRequest) {
    21→  try {
    22→    const { shows, title, subtitle, organization } = await request.json();
    23→
    24→    if (!shows || !Array.isArray(shows) || shows.length === 0) {
    25→      return NextResponse.json({ error: 'No shows provided' }, { status: 400 });
    26→    }
    27→
    28→    // Build the base URL so relative image paths (e.g. /program-images/p1_img1.jpeg)
    29→    // resolve correctly when the HTML is written into a print-to-PDF window
    30→    // (which has origin 'about:blank'). Fall back to the request origin.
    31→    const origin = request.nextUrl.origin;
    32→
    33→    // Generate a professional multi-section HTML report
    34→    const html = generatePDFHtml(shows, {
    35→      title: title || 'گزارش جامع آرشیو برنامه‌های تلویزیونی',
    36→      subtitle: subtitle || 'بانک رسانه — آرشیو تخصصی برنامه‌های تلویزیونی',
    37→      organization: organization || 'بانک رسانه',
    38→      origin,
    39→    });
    40→
    41→    return new NextResponse(html, {
    42→      status: 200,
    43→      headers: {
    44→        'Content-Type': 'text/html; charset=utf-8',
    45→        'Content-Disposition': `attachment; filename="banke-rasane-report-${Date.now()}.html"`,
    46→      },
    47→    });
    48→  } catch (error) {
    49→    console.error('PDF export error:', error);
    50→    return NextResponse.json(
    51→      { error: 'PDF export failed', details: error instanceof Error ? error.message : String(error) },
    52→      { status: 500 }
    53→    );
    54→  }
    55→}

    69→  const total = shows.length;
    70→
    71→  // ---------- Aggregations ----------
    72→  // Group by category
    73→  const byCategory: Record<string, any[]> = {};
    74→  shows.forEach(s => {
    75→    const cat = (s.category || 'سایر').replace(/^\d+\.\s*/, '').trim() || 'سایر';

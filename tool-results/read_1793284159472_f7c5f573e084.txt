    42→// A "section-specific tag" — derives a small badge label from a TalkShow field
    43→// (only when the field has a meaningful value). Each Genre chooses which tags to show.
    44→type TagSpec = { id: string; label: (s: TalkShow) => string | null };
    45→
    46→interface Genre {
    47→  id: string;
    48→  key: string;
    49→  nameFa: string;
    50→  name: string;          // English name (also serves as `nameEn`)
    51→  nameEn?: string;       // Optional explicit English name (defaults to `name`)
    52→  color: string;
    53→  description: string;   // Persian description
    54→  descriptionEn?: string; // Optional English description
    55→  filterFn: (s: TalkShow) => boolean;
    56→  // Section-specific tags shown on cards in this genre's view
    57→  tags?: TagSpec[];
    58→  // Preset tags offered as quick-add buttons in AddEditView when this genre is preset
    59→  presetTags?: string[];
    60→  // Which TalkShow fields the ProgramsView search bar should match against
    61→  searchFields?: (keyof TalkShow)[];
    62→  // True for user-created genres (stored in localStorage)
    63→  isCustom?: boolean;
    64→}
    65→
    66→// Returns the localized name of a genre based on the current language.
    67→// Falls back to `name` (English) if `nameEn` is not set, and to `nameFa` for Farsi.
    68→function genreName(g: Genre, lang: Lang): string {
    69→  if (lang === 'en') return g.nameEn || g.name || g.nameFa;
    70→  return g.nameFa || g.name;
    71→}
    72→
    73→// Returns the localized description of a genre (English version optional).
    74→function genreDescription(g: Genre, lang: Lang): string {
    75→  if (lang === 'en') return g.descriptionEn || g.description;
    76→  return g.description;
    77→}

    79→// Small inline component that pulls a translated string from the LangContext.
    80→// Useful when a component can't call useLang() at the top of its body
    81→// (e.g. inside a JSX prop that needs translation).
    82→function LangText({ k, args, style, className }: { k: string; args?: (string | number)[]; style?: React.CSSProperties; className?: string }) {
    83→  const { t } = useLang();
    84→  return <span style={style} className={className}>{t(k, args)}</span>;
    85→}

   390→// Returns the list of Islamic countries mentioned in a show's text (description, summary, country).
   391→function detectCountries(show: TalkShow): string[] {
   392→  const text = `${show.descriptionFa || ''} ${show.summary || ''} ${show.country || ''}`;
   393→  const found: string[] = [];
   394→  for (const c of ISLAMIC_COUNTRIES) {
   395→    // Word-boundary-style match: country must not be part of a longer Persian word.
   396→    // We use a regex that requires non-letter / start / end on either side.
   397→    const escaped = c.replace(/[.*+?^${}()|[\]\\]/g, '\\$&');
   398→    const re = new RegExp(`(^|[^\\u0600-\\u06FFa-zA-Z])${escaped}([^\\u0600-\\u06FFa-zA-Z]|$)`);
   399→    if (re.test(text)) found.push(c);
   400→  }
   401→  return Array.from(new Set(found));
   402→}

   404→// ============================================
   405→// RELATED PROGRAMS
   406→// Score: same category +5, same layout +3, shared elements (Jaccard up to +3),
   407→//        same style +2, same effects +2, same country +2, same size +1, same viewers +1
   408→// ============================================
   409→function findRelated(show: TalkShow, allShows: TalkShow[], limit = 6): TalkShow[] {
   410→  const showElements = (show.elements || '').split(/[,،\s]+/).filter(Boolean);
   411→  const candidates = allShows.filter(s => s.num !== show.num);
   412→  const scored = candidates.map(s => {
   413→    let score = 0;
   414→    if (s.category === show.category) score += 5;
   415→    if (s.layout === show.layout) score += 3;
   416→    const sElements = (s.elements || '').split(/[,،\s]+/).filter(Boolean);
   417→    if (showElements.length > 0 && sElements.length > 0) {
   418→      const intersection = showElements.filter(e => sElements.includes(e)).length;
   419→      const union = new Set([...showElements, ...sElements]).size || 1;
   420→      score += Math.round((intersection / union) * 3);
   421→    }
   422→    if (s.style === show.style) score += 2;
   423→    if (s.effects === show.effects) score += 2;
   424→    if (s.country && show.country && s.country === show.country) score += 2;
   425→    if (s.size === show.size) score += 1;
   426→    if (s.viewers === show.viewers) score += 1;
   427→    return { show: s, score };
   428→  });
   429→  return scored
   430→    .filter(x => x.score > 0)
   431→    .sort((a, b) => b.score - a.score)
   432→    .slice(0, limit)
   433→    .map(x => x.show);
   434→}

   505→// Adjust a hex color brightness (-100 to +100)
   506→function adjustColor(hex: string, amount: number): string {
   507→  let r = parseInt(hex.slice(1, 3), 16);
   508→  let g = parseInt(hex.slice(3, 5), 16);
   509→  let b = parseInt(hex.slice(5, 7), 16);
   510→  r = Math.max(0, Math.min(255, r + amount));
   511→  g = Math.max(0, Math.min(255, g + amount));
   512→  b = Math.max(0, Math.min(255, b + amount));
   513→  return `#${r.toString(16).padStart(2, '0')}${g.toString(16).padStart(2, '0')}${b.toString(16).padStart(2, '0')}`;
   514→}

   698→type Overrides = {
   699→  added: TalkShow[];                       // shows created by the user
   700→  edited: Record<number, TalkShow>;        // edits to shows from the base JSON
   701→  deleted: number[];                       // base shows hidden by the user
   702→};

   704→const OVERRIDES_KEY = 'bankresaneh_overrides_v1';
   705→
   706→function loadOverrides(): Overrides {
   707→  if (typeof window === 'undefined') return { added: [], edited: {}, deleted: [] };
   708→  try {
   709→    const raw = window.localStorage.getItem(OVERRIDES_KEY);
   710→    if (!raw) return { added: [], edited: {}, deleted: [] };
   711→    const o = JSON.parse(raw);
   712→    return {
   713→      added: Array.isArray(o.added) ? o.added : [],
   714→      edited: o.edited && typeof o.edited === 'object' ? o.edited : {},
   715→      deleted: Array.isArray(o.deleted) ? o.deleted : [],
   716→    };
   717→  } catch {
   718→    return { added: [], edited: {}, deleted: [] };
   719→  }
   720→}
   721→
   722→function saveOverrides(o: Overrides) {
   723→  if (typeof window === 'undefined') return;
   724→  try {
   725→    window.localStorage.setItem(OVERRIDES_KEY, JSON.stringify(o));
   726→  } catch {
   727→    // Storage full or unavailable — silently ignore (edits stay in-memory only).
   728→  }
   729→}

   731→// Merge base shows with the user's overrides: drop deleted ones, replace edited ones, prepend added ones.
   732→function applyOverrides(base: TalkShow[], o: Overrides): TalkShow[] {
   733→  const deleted = new Set(o.deleted);
   734→  const out: TalkShow[] = [];
   735→  for (const s of base) {
   736→    if (deleted.has(s.num)) continue;
   737→    out.push(o.edited[s.num] || s);
   738→  }
   739→  return [...o.added, ...out];
   740→}

     1→import { NextRequest, NextResponse } from 'next/server';
     2→
     3→export const runtime = 'nodejs';
     4→
     5→export async function POST(request: NextRequest) {
     6→  try {
     7→    const { shows, title, subtitle, organization } = await request.json();
     8→
     9→    if (!shows || !Array.isArray(shows) || shows.length === 0) {
    10→      return NextResponse.json({ error: 'No shows provided' }, { status: 400 });
    11→    }
    12→
    13→    // Generate a professional multi-section HTML report
    14→    const html = generatePDFHtml(shows, {
    15→      title: title || 'گزارش جامع آرشیو برنامه‌های تلویزیونی',
    16→      subtitle: subtitle || 'بانک رسانه — آرشیو تخصصی برنامه‌های تلویزیونی',
    17→      organization: organization || 'بانک رسانه',
    18→    });
    19→
    20→    return new NextResponse(html, {
    21→      status: 200,
    22→      headers: {
    23→        'Content-Type': 'text/html; charset=utf-8',
    24→        'Content-Disposition': `attachment; filename="banke-rasane-report-${Date.now()}.html"`,
    25→      },
    26→    });
    27→  } catch (error) {
    28→    console.error('PDF export error:', error);
    29→    return NextResponse.json(
    30→      { error: 'PDF export failed', details: error instanceof Error ? error.message : String(error) },
    31→      { status: 500 }
    32→    );
    33→  }
    34→}

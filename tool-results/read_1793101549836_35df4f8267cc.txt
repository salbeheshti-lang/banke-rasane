    70→    70→// Adjust a hex color brightness (-100 to +100)
    71→    71→function adjustColor(hex: string, amount: number): string {
    72→    72→  let r = parseInt(hex.slice(1, 3), 16);
    73→    73→  let g = parseInt(hex.slice(3, 5), 16);
    74→    74→  let b = parseInt(hex.slice(5, 7), 16);
    75→    75→  r = Math.max(0, Math.min(255, r + amount));
    76→    76→  g = Math.max(0, Math.min(255, g + amount));
    77→    77→  b = Math.max(0, Math.min(255, b + amount));
    78→    78→  return `#${r.toString(16).padStart(2, '0')}${g.toString(16).padStart(2, '0')}${b.toString(16).padStart(2, '0')}`;
    79→    79→}

   261→   261→// LOCAL PERSISTENCE — user edits are saved in localStorage so they survive a refresh.
   262→   262→// ============================================
   263→   263→type Overrides = {
   264→   264→  added: TalkShow[];                       // shows created by the user
   265→   265→  edited: Record<number, TalkShow>;        // edits to shows from the base JSON
   266→   266→  deleted: number[];                       // base shows hidden by the user
   267→   267→};
   268→   268→
   269→   269→const OVERRIDES_KEY = 'bankresaneh_overrides_v1';
   270→   270→
   271→   271→function loadOverrides(): Overrides {
   272→   272→  if (typeof window === 'undefined') return { added: [], edited: {}, deleted: [] };
   273→   273→  try {
   274→   274→    const raw = window.localStorage.getItem(OVERRIDES_KEY);
   275→   275→    if (!raw) return { added: [], edited: {}, deleted: [] };
   276→   276→    const o = JSON.parse(raw);
   277→   277→    return {
   278→   278→      added: Array.isArray(o.added) ? o.added : [],
   279→   279→      edited: o.edited && typeof o.edited === 'object' ? o.edited : {},
   280→   280→      deleted: Array.isArray(o.deleted) ? o.deleted : [],
   281→   281→    };
   282→   282→  } catch {
   283→   283→    return { added: [], edited: {}, deleted: [] };
   284→   284→  }
   285→   285→}
   286→   286→
   287→   287→function saveOverrides(o: Overrides) {
   288→   288→  if (typeof window === 'undefined') return;
   289→   289→  try {
   290→   290→    window.localStorage.setItem(OVERRIDES_KEY, JSON.stringify(o));
   291→   291→  } catch {
   292→   292→    // Storage full or unavailable — silently ignore (edits stay in-memory only).
   293→   293→  }
   294→   294→}
   295→   295→
   296→   296→// Merge base shows with the user's overrides: drop deleted ones, replace edited ones, prepend added ones.
   297→   297→function applyOverrides(base: TalkShow[], o: Overrides): TalkShow[] {
   298→   298→  const deleted = new Set(o.deleted);
   299→   299→  const out: TalkShow[] = [];
   300→   300→  for (const s of base) {
   301→   301→    if (deleted.has(s.num)) continue;
   302→   302→    out.push(o.edited[s.num] || s);
   303→   303→  }
   304→   304→  return [...o.added, ...out];
   305→   305→}

    20→    20→export async function POST(request: NextRequest) {
    21→    21→  try {
    22→    22→    const { shows, title, subtitle, organization } = await request.json();
    23→    23→
    24→    24→    if (!shows || !Array.isArray(shows) || shows.length === 0) {
    25→    25→      return NextResponse.json({ error: 'No shows provided' }, { status: 400 });
    26→    26→    }
    27→    27→
    28→    28→    // Build the base URL so relative image paths (e.g. /program-images/p1_img1.jpeg)
    29→    29→    // resolve correctly when the HTML is written into a print-to-PDF window
    30→    30→    // (which has origin 'about:blank'). Fall back to the request origin.
    31→    31→    const origin = request.nextUrl.origin;
    32→    32→
    33→    33→    // Generate a professional multi-section HTML report
    34→    34→    const html = generatePDFHtml(shows, {
    35→    35→      title: title || 'گزارش جامع آرشیو برنامه‌های تلویزیونی',
    36→    36→      subtitle: subtitle || 'بانک رسانه — آرشیو تخصصی برنامه‌های تلویزیونی',
    37→    37→      organization: organization || 'بانک رسانه',
    38→    38→      origin,
    39→    39→    });
    40→    40→
    41→    41→    return new NextResponse(html, {
    42→    42→      status: 200,
    43→    43→      headers: {
    44→    44→        'Content-Type': 'text/html; charset=utf-8',
    45→    45→        'Content-Disposition': `attachment; filename="banke-rasane-report-${Date.now()}.html"`,
    46→    46→      },
    47→    47→    });
    48→    48→  } catch (error) {
    49→    49→    console.error('PDF export error:', error);
    50→    50→    return NextResponse.json(
    51→    51→      { error: 'PDF export failed', details: error instanceof Error ? error.message : String(error) },
    52→    52→      { status: 500 }
    53→    53→    );
    54→    54→  }
    55→    55→}
